package list

import (
	"strings"

	"github.com/bostroemc/tui/opcua-browser/types"
	tea "github.com/charmbracelet/bubbletea"
	"github.com/gopcua/opcua/ua"
)

func New(read chan types.OpcUaReadData, root string) Model {
	node, _ := ua.ParseNodeID(root)
	return Model{read: read, Styles: types.DefaultStyles(), Node: node, Path: root} // "ns=8;s=plc/app/Application/sym"}
}

type Model struct {
	index    int
	read     chan types.OpcUaReadData
	Path     string
	Active   int
	Node     *ua.NodeID
	Parent   types.Node
	Children []types.Node
	id       int
	max      int // (min, max) chosen to obey these rules: 0<=min<=max<=len(Children); min<= index<= max; max-min <= Height
	min      int
	Height   int
	Width    int
	Styles   types.Styles
}

func (m Model) Init() tea.Cmd {
	return nil //tea.Batch(browseCmd(m.browse), initCmd(m.browse))
}

func (m Model) Update(msg tea.Msg) (Model, tea.Cmd) {
	switch msg := msg.(type) {

	case tea.KeyMsg:
		if msg.String() == "j" || msg.String() == "down" {
			m.index++
			m.index = min(m.index, len(m.Children)-1)
			m.SetMinMax(m.min, m.max)
		}
		if msg.String() == "k" || msg.String() == "up" {
			m.index--
			m.index = max(m.index, 0)
			m.SetMinMax(m.min, m.max)
		}
	}
	return m, nil
}

func (m Model) View() string {
	if len(m.Children) == 0 {
	}
	var s strings.Builder
	s.WriteString(m.Parent.BrowseName + "\n")
	for i, c := range m.Children {
		if i < m.min || i > m.max {
			continue
		}
		if m.index == i {
			s.WriteString("  " + m.Styles.Index.Render(c.BrowseName) + m.Styles.Index.Render(c.DataType) + "\n")
		} else {
			s.WriteString("  " + c.BrowseName + c.DataType + "\n")
		}

	}
	// s.WriteString(m.stack.Peek() + " " + strconv.Itoa(m.min) + " " + strconv.Itoa(m.max) + " " + strconv.Itoa(m.Height) + "\n")

	m.Styles.Body = m.Styles.Body.Width(m.Width).Height(m.Height)
	return m.Styles.Body.Render(s.String())
}

func (m *Model) SetView(height, width int) {
	m.Height, m.Width = height, width
	if m.max > m.Height-1 || m.max == m.min {
		m.max = m.min + m.Height - 1 //add that m.max must not be greater then len(Children)
		// m.max = min(m.max, len(m.Children))
	}
}

func (m *Model) SetMinMax(minimum, maximum int) {
	if m.index < minimum {
		m.min = m.index
		m.max = m.index + m.Height - 1
		m.max = min(m.max, len(m.Children)-1)
	}
	if m.index > maximum {
		m.max = m.index
		m.min = m.index - m.Height + 1
		m.min = max(m.min, 0)
	}
}
